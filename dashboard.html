C-1 — Monthly Revenue Trend (CTE)
CTE
WITH monthly_revenue AS ( SELECT DATE_TRUNC('month', o.order_date) AS month, SUM(oi.quantity * oi.unit_price * (1 - oi.discount_pct/100.0)) AS net_revenue FROM orders o JOIN order_items oi ON oi.order_id = o.order_id WHERE o.status NOT IN ('Cancelled', 'Returned') GROUP BY 1 ) SELECT month, ROUND(net_revenue, 2) AS monthly_revenue, ROUND(SUM(net_revenue) OVER (ORDER BY month), 2) AS running_total FROM monthly_revenue ORDER BY month;

C-3 — Customer Lifetime Value (4-CTE Chain)
MULTI-CTE
WITH order_totals AS ( -- Step 1: revenue per order SELECT o.customer_id, o.order_id, SUM(oi.quantity * oi.unit_price * (1-oi.discount_pct/100)) AS order_value FROM orders o JOIN order_items oi ON oi.order_id=o.order_id GROUP BY 1,2 ), customer_stats AS ( -- Step 2: aggregate per customer SELECT customer_id, COUNT(order_id) orders, SUM(order_value) lifetime_value, AVG(order_value) avg_order FROM order_totals GROUP BY 1 ), clv_scored AS ( -- Step 3: score and tier SELECT *, NTILE(4) OVER (ORDER BY lifetime_value) AS clv_tier FROM customer_stats ) SELECT c.first_name||' '||c.last_name customer, cs.lifetime_value, cs.orders, cs.avg_order, CASE cs.clv_tier WHEN 4 THEN 'High Value' WHEN 3 THEN 'Medium' ELSE 'Low Value' END AS clv_segment FROM clv_scored cs JOIN customers c ON c.customer_id=cs.customer_id ORDER BY lifetime_value DESC LIMIT 10;

❌ SLOW — Correlated Subquery (O(n²))
SELECT c.customer_id, c.first_name,
  (SELECT SUM(oi2.quantity * oi2.unit_price)
   FROM   orders o2
   JOIN   order_items oi2 ON oi2.order_id=o2.order_id
   WHERE  o2.customer_id=c.customer_id
     AND  o2.status='Delivered') AS total_spend
FROM   customers c
ORDER BY total_spend DESC LIMIT 10;
-- Fires subquery once per customer row
-- ~2,000 × 18,000 = 36M comparisons
✅ FAST — Pre-aggregate with CTE
WITH spend_agg AS (
    SELECT o.customer_id,
           SUM(oi.quantity*oi.unit_price) AS total_spend
    FROM   orders      o
    JOIN   order_items oi ON oi.order_id=o.order_id
    WHERE  o.status='Delivered'
    GROUP BY o.customer_id
)
SELECT c.customer_id, c.first_name, sa.total_spend
FROM   customers c
JOIN   spend_agg sa ON sa.customer_id=c.customer_id
ORDER BY sa.total_spend DESC LIMIT 10;
-- Aggregates once, joins once — O(n)
